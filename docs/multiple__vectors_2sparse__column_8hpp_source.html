44void multiply_sparse_column_with_multiple_vectors_internal(
45 const tatami::Matrix<LeftValue_, LeftIndex_>& left,
46 const LeftIndex_ start,
47 const LeftIndex_ length,
48 const LeftIndex_ left_NR,
49 const RightIndex_ right_NC,
50 GetRight_ get_right,
51 GetOutput_ get_output,
52 const MultiplySparseColumnWithMultipleVectorsOptions& options

54 auto ext = tatami::consecutive_extractor<true>(left, false, start, length);
55 typedef I<decltype(get_output(0)[0])> Output;
56
57 if (options.block_size == 1) {
58 auto vbuffer = tatami::create_container_of_Index_size<std::vector<LeftValue_> >(left_NR);
59 auto ibuffer = tatami::create_container_of_Index_size<std::vector<LeftIndex_> >(left_NR);
60 for (LeftIndex_ cd = 0; cd < length; ++cd) {
61 const auto range = ext->fetch(vbuffer.data(), ibuffer.data());
62 if (range.number == 0) {
63 continue;
64 }
65

67 const auto optr = get_output(rc);
68 const Output mult = get_right(rc)[start + cd];
69 for (LeftIndex_ x = 0; x < range.number; ++x) {
70 optr[range.index[x]] += mult * static_cast<Output>(range.value[x]);
71 }
72 }
73 }
74
75 } else {

77 std::vector<std::vector<LeftIndex_> > left_ibuffers;
78 std::vector<tatami::SparseRange<LeftValue_, LeftIndex_> > left_ranges;
79 std::vector<LeftIndex_> left_non_empty;
80 {
81 const LeftIndex_ max_block_cols = sanisizer::min(length, options.block_size);
82 left_vbuffers.reserve(max_block_cols);
83 left_ibuffers.reserve(max_block_cols);
84 for (LeftIndex_ cd = 0; cd < max_block_cols; ++cd) {
85 left_vbuffers.emplace_back(tatami::cast_Index_to_container_size<std::vector<LeftValue_> >(left_NR));
86 left_ibuffers.emplace_back(tatami::cast_Index_to_container_size<std::vector<LeftIndex_> >(left_NR));
87 }
88 sanisizer::resize(left_ranges, max_block_cols);
89 left_non_empty.reserve(max_block_cols);
90 }
91
92 LeftIndex_ cd = 0;
93 while (cd < length) {
94 // Only considering the LHS columns with at least one structural non-zero.
95 // Thus, our block consists of 'options.block_size' non-empty LHS columns, rather than fixed column-wise chunks of the LHS matrix.
96 // This ensures that we don't waste iterations on LHS columns that don't contribute anything to the product.
97 const auto left_block_info = fetch_non_empty_sparse_block(
98 *ext,
99 left_vbuffers,
100 left_ibuffers,
101 left_ranges,
102 left_non_empty,
103 cd,
104 length,
105 options.block_size,
106 /* zero = */ [](LeftIndex_) -> void {} // No need to worry about zeroing as the buffers have already been zeroed.
107 );
108 const auto cd_num = left_block_info.num_non_empty;
109
110 // If the LHS columns are all non-empty, we can speed up the loops by just using a simple counter to get the column indices.
111 // Otherwise, we'll have to access the 'left_non_empty' vector to figure out the indices of each non-empty column.
112 if (left_block_info.all_non_empty) {
113 const LeftIndex_ cd_base = start + cd;
114 for (RightIndex_ rc = 0; rc < right_NC; ++rc) {
115 const auto outcol = get_output(rc);
116 const auto rightcol = get_right(rc);
117 for (LeftIndex_ cd_counter = 0; cd_counter < cd_num; ++cd_counter) {
118 const auto& currange = left_ranges[cd_counter];
119 const Output mult = rightcol[cd_base + cd_counter];
120 for (LeftIndex_ x = 0; x < currange.number; ++x) {
121 outcol[currange.index[x]] += mult * static_cast<Output>(currange.value[x]);
122 }
123 }
124 }
125
126 } else {
127 for (auto& cdne : left_non_empty) {
128 cdne += start;
129 }
130 for (RightIndex_ rc = 0; rc < right_NC; ++rc) {
131 const auto outcol = get_output(rc);
132 const auto rightcol = get_right(rc);
133 for (LeftIndex_ cd_counter = 0; cd_counter < cd_num; ++cd_counter) {
134 const auto& currange = left_ranges[cd_counter];
135 const Output mult = rightcol[left_non_empty[cd_counter]];
136 for (LeftIndex_ x = 0; x < currange.number; ++x) {
137 outcol[currange.index[x]] += mult * static_cast<Output>(currange.value[x]);
138 }
139 }
140 }
141 }
142
143 cd = left_block_info.position;
144 }
145 }
